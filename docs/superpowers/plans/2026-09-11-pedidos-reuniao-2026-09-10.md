- [ ] **Step 1: Escrever a função de espera do PDF em `lib/api.ts`**

Perto de `fetchFiscalNotaPdfUrl`, adicionar:

```ts
// Reunião 2026-09-10 (min 14:34): ao fechar a venda, o app só imprimia o
// comprovante SEM valor fiscal — a nota autorizada ficava só em
// Administração → Notas Fiscais. O dono confirmou que, quando a venda emite
// nota, o que deve sair na impressora é o cupom fiscal.
//
// A emissão é assíncrona (triggerEmissaoFiscal, fire-and-forget) e passa por
// SEFAZ, então o PDF não existe no instante do fechamento. Esta função
// espera em poll curto, com teto — nunca trava o caixa: estourou o tempo,
// devolve null e quem chamou cai no comprovante normal.
export async function aguardarNotaFiscalDaVenda(
  params: { orderId?: string; tableId?: string },
  opts: { timeoutMs?: number; intervalMs?: number } = {},
): Promise<{ pdfUrl: string } | null> {
  const timeoutMs = opts.timeoutMs ?? 12000;
  const intervalMs = opts.intervalMs ?? 1500;
  const limite = Date.now() + timeoutMs;

  while (Date.now() < limite) {
    try {
      const { data } = await supabase
        .from('fiscal_notas')
        .select('id, status')
        // Só a nota do fechamento inteiro (pessoa_identificador null) — nota
        // individual por pessoa (migration 055) já é impressa no próprio
        // fluxo "Por Cliente", não deve reabrir aqui.
        .is('pessoa_identificador', null)
        .eq(params.orderId ? 'order_id' : 'table_id', params.orderId ?? params.tableId!)
        .order('created_at', { ascending: false })
        .limit(1);

      const nota = data?.[0];
      if (nota?.status === 'autorizada') {
        const pdfUrl = await fetchFiscalNotaPdfUrl(nota.id);
        if (pdfUrl) return { pdfUrl };
      }
      if (nota?.status === 'erro') return null; // rejeitada: não adianta esperar
    } catch {
      // rede instável não pode travar o fechamento — tenta de novo até o teto
    }
    await new Promise((r) => setTimeout(r, intervalMs));
  }
  return null;
}
```
